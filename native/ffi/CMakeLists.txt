cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LuiTrustedFfi LANGUAGES C CXX)
endif()
option(LUI_FFI_BROKER "Build the optional native call broker" OFF)
add_subdirectory(dependency)
if(NOT TARGET nlohmann_json::nlohmann_json)
    include(FetchContent)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/Dependencies.json" Pins)
    string(JSON JsonUrl GET "${Pins}" JsonUrl)
    string(JSON JsonHash GET "${Pins}" JsonSha256)
    set(JSON_BuildTests OFF)
    set(JSON_Install OFF)
    FetchContent_Declare(LuiJson URL "${JsonUrl}" URL_HASH "SHA256=${JsonHash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(LuiJson)
endif()
find_package(Threads REQUIRED)
add_library(LuiFfiCore STATIC Core.cpp Types.cpp Images.cpp Callbacks.cpp)
set_target_properties(LuiFfiCore PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_features(LuiFfiCore PUBLIC cxx_std_17)
target_include_directories(LuiFfiCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../abi")
target_link_libraries(LuiFfiCore PUBLIC LuiLibffi nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
add_library(LuiFfi MODULE Provider.cpp)
set_target_properties(LuiFfi PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_definitions(LuiFfi PRIVATE LUI_EXTENSION_BUILD)
target_link_libraries(LuiFfi PRIVATE LuiFfiCore)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(Exports "${CMAKE_CURRENT_SOURCE_DIR}/Exports.map")
    target_link_options(LuiFfi PRIVATE "LINKER:--version-script=${Exports}")
    set_property(TARGET LuiFfi APPEND PROPERTY LINK_DEPENDS "${Exports}")
endif()
if(LUI_FFI_BROKER)
    add_executable(LuiFfiBroker Broker.cpp)
    target_link_libraries(LuiFfiBroker PRIVATE LuiFfiCore)
endif()
set(NoticeDir "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/licenses")
foreach(Artifact LuiFfi LuiFfiBroker)
    if(TARGET ${Artifact})
        add_custom_command(TARGET ${Artifact} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:${Artifact}>/THIRD-PARTY-LICENSES"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${NoticeDir}/Libffi.txt" "${NoticeDir}/Json.txt" "${NoticeDir}/PicoSHA2.txt"
                "$<TARGET_FILE_DIR:${Artifact}>/THIRD-PARTY-LICENSES" VERBATIM)
    endif()
endforeach()
