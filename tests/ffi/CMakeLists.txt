cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LuiFfiQualification LANGUAGES C CXX)
    enable_testing()
endif()

find_package(Threads REQUIRED)
add_library(LuiFfiFixture SHARED Fixture.cpp)
target_compile_definitions(LuiFfiFixture PRIVATE LUI_BUILD_FFI_FIXTURE)
target_compile_features(LuiFfiFixture PRIVATE cxx_std_17)
target_link_libraries(LuiFfiFixture PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(LuiFfiFixture PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

add_executable(LuiFfiOracle Oracle.c)
set_target_properties(LuiFfiOracle PROPERTIES BUILD_RPATH_USE_ORIGIN YES)
target_compile_features(LuiFfiOracle PRIVATE c_std_11)
target_link_libraries(LuiFfiOracle PRIVATE LuiFfiFixture Threads::Threads)
add_test(NAME LuiFfiNativeOracle COMMAND LuiFfiOracle)
set_tests_properties(LuiFfiNativeOracle PROPERTIES TIMEOUT 8)

if(TARGET LuiFfiCore)
    add_executable(LuiFfiCallCosts EXCLUDE_FROM_ALL CallCosts.cpp)
    target_link_libraries(LuiFfiCallCosts PRIVATE LuiFfiCore LuiFfiFixture)
    set_target_properties(LuiFfiCallCosts PROPERTIES BUILD_RPATH_USE_ORIGIN YES)
endif()
