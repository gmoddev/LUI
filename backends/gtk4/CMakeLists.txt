if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "[LUI:GTK] This profile requires Linux")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(LuiGtk REQUIRED IMPORTED_TARGET gtk4>=4.10)
add_library(LuiGtkBackend STATIC Backend.cpp Geometry.cpp Platform.cpp Images.cpp)
target_include_directories(LuiGtkBackend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(LuiGtkBackend PUBLIC LuiRuntime PkgConfig::LuiGtk PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(LuiGtkBackend PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_executable(LuiGtkHost Main.cpp)
target_link_libraries(LuiGtkHost PRIVATE LuiGtkBackend LuiProject nlohmann_json::nlohmann_json)
set_target_properties(LuiGtkHost PROPERTIES OUTPUT_NAME lui-gtk)
set_target_properties(LuiGtkHost PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS LuiGtkHost RUNTIME DESTINATION bin COMPONENT GtkHost)
install(TARGETS LuiRuntime LIBRARY DESTINATION lib COMPONENT GtkHost)
install(FILES "${PROJECT_SOURCE_DIR}/LICENSE" DESTINATION . COMPONENT GtkHost)
install(FILES "${LUI_LUAU_LICENSE}" DESTINATION THIRD-PARTY-LICENSES RENAME Luau.txt COMPONENT GtkHost)
install(FILES "${asio_SOURCE_DIR}/asio/LICENSE_1_0.txt" DESTINATION THIRD-PARTY-LICENSES RENAME Asio.txt COMPONENT GtkHost)
install(FILES "${PROJECT_SOURCE_DIR}/third_party/licenses/Json.txt" "${PROJECT_SOURCE_DIR}/third_party/licenses/OpenSSL.txt" DESTINATION THIRD-PARTY-LICENSES COMPONENT GtkHost)
add_executable(LuiGtkTests "${PROJECT_SOURCE_DIR}/tests/gtk4/Backend.cpp")
target_compile_definitions(LuiGtkTests PRIVATE LUI_CONVERSATION_DIR="${PROJECT_SOURCE_DIR}/examples/conversation")
target_link_libraries(LuiGtkTests PRIVATE LuiGtkBackend LuiProject nlohmann_json::nlohmann_json)
target_compile_definitions(LuiGtkTests PRIVATE LUI_MODULAR_MANIFEST="${PROJECT_SOURCE_DIR}/examples/modular/lui.json")
find_program(LuiXvfb xvfb-run REQUIRED)
find_program(LuiDbus dbus-run-session REQUIRED)
add_test(NAME LuiGtkTests COMMAND "${LuiXvfb}" -a "${LuiDbus}" -- "$<TARGET_FILE:LuiGtkTests>")
set_tests_properties(LuiGtkTests PROPERTIES TIMEOUT 45 ENVIRONMENT "GDK_BACKEND=x11;GSK_RENDERER=cairo;GTK_A11Y=none;GTK_USE_PORTAL=0;G_DEBUG=fatal-criticals")
add_executable(LuiGtkTemporaryTextTests "${PROJECT_SOURCE_DIR}/tests/gtk4/TemporaryText.cpp")
target_link_libraries(LuiGtkTemporaryTextTests PRIVATE PkgConfig::LuiGtk)
target_compile_options(LuiGtkTemporaryTextTests PRIVATE -Wall -Wextra -Werror)
add_test(NAME LuiGtkTemporaryTextTests COMMAND "${LuiXvfb}" -a "$<TARGET_FILE:LuiGtkTemporaryTextTests>")
set_tests_properties(LuiGtkTemporaryTextTests PROPERTIES TIMEOUT 30 ENVIRONMENT "GDK_BACKEND=x11;GSK_RENDERER=cairo;GTK_A11Y=none;G_DEBUG=fatal-criticals")
add_custom_target(LuiGtkQualification DEPENDS LuiGtkHost LuiGtkTests LuiGtkTemporaryTextTests LuiProjectTool)
